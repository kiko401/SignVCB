# 1. 停止并重建所有容器
docker-compose -f deploy/docker-compose.yml --env-file .env down
docker-compose -f deploy/docker-compose.yml --env-file .env up -d --build

# 2. 确认服务状态
docker-compose -f deploy/docker-compose.yml ps
# 等待 MySQL 和 Redis 变成 healthy 状态（几十秒）

# 3. 初始化数据库表结构
docker-compose -f deploy/docker-compose.yml exec signvcb-server alembic upgrade head

# 4. 导入初始数据（读物 + 练习题）
docker-compose -f deploy/docker-compose.yml exec signvcb-server python -m app.scripts.seed_local

# 5. 注入 ONNX 模型到 /app/models/
docker cp Offline-Tools/outputs/onnx/csl_encoder.onnx signvcb-engine:/app/models/

# 6. 注入 tokenizer 到 /app/data/（ONNXEncoder 从这里加载 tokenizer）
docker cp Offline-Tools/outputs/onnx/tokenizer_config.json signvcb-engine:/app/data/
docker cp Offline-Tools/outputs/onnx/tokenizer.json signvcb-engine:/app/data/
docker cp Offline-Tools/outputs/onnx/vocab.txt signvcb-engine:/app/data/
docker cp Offline-Tools/outputs/onnx/config.json signvcb-engine:/app/data/
docker cp Offline-Tools/outputs/onnx/special_tokens_map.json signvcb-engine:/app/data/

# 7. 注入词表、fallback词典、FAISS索引到 /app/data/
docker cp Offline-Tools/outputs/data/csl_standard_vocab.json signvcb-engine:/app/data/vocab.json
docker cp Offline-Tools/outputs/fallback/initial_fallback.json signvcb-engine:/app/data/fallback/initial_fallback.json
docker cp Offline-Tools/outputs/faiss/faiss_index.bin signvcb-engine:/app/data/faiss/faiss_index.bin

# 8. 重启引擎使新模型生效
docker-compose -f deploy/docker-compose.yml restart signvcb-engine

# 9. 确认引擎健康
docker-compose -f deploy/docker-compose.yml ps
