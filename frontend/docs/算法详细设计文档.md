## 第一部分：模型选型与工程化基建

本项目不涉及大模型微调，核心在于"轻量级向量模型的本地工程化加速"与"大模型 API 的结构化调度"。

### 1. 核心模型选型与转换（本地预处理）

- **大语言模型 (LLM)**：`DeepSeek-V4`（API 模型名 `deepseek-v4-flash`）。无需本地部署，通过 `httpx` 异步调用，强制开启 `response_format={"type": "json_object"}`。
    - **API 端点**：`https://api.deepseek.com/v1/chat/completions`
    - **超时策略**：连接超时 2s，读取超时 4s（v2.1 收紧，留出后端 10s 整体超时的余量）。
    - **容错**：超时或异常时，算法引擎不下发 `refined_pass`，由后端补发 `fallback` 事件。

- **向量化模型**：`BAAI/bge-small-zh-v1.5`（768 维）。
    - **选型理由**：体积小（约 100MB），中文语义强，适合云端 4C4G CPU 环境推理。
    - **ONNX 转换（离线任务）**：算法工程师需在本地（含 GPU 环境）使用 `optimum` 库将 PyTorch 模型转换为 ONNX 格式，并使用 `onnxruntime` 进行 INT8 量化，产出 `csl_encoder.onnx` 与 `config.json`。
    - **工程约束**：云端严禁加载 PyTorch 原生权重，必须使用 ONNX 运行时以降低内存占用并提升 3 倍以上推理速度。
    - **v3.0 路径统一**：模型文件统一存放于容器内 `/app/models/csl_encoder.onnx`，FAISS 索引存放于 `/app/data/faiss_index.bin`，标准词表存放于 `/app/data/vocab.json`。

### 1. ONNX 模型转换与 INT8 量化

```python
# scripts/convert_onnx.py
"""
离线脚本：将 BAAI/bge-small-zh-v1.5 PyTorch 模型转换为 ONNX INT8 量化模型。
运行环境：本地 GPU 环境（无需云端执行）。
产出文件：models/csl_encoder.onnx
"""
import os
from optimum.onnxruntime import ORTModelForFeatureExtraction
from transformers import AutoTokenizer
from onnxruntime.quantization import quantize_dynamic, QuantType


MODEL_ID = "BAAI/bge-small-zh-v1.5"
OUTPUT_DIR = "models"
ONNX_PATH = os.path.join(OUTPUT_DIR, "csl_encoder.onnx")
QUANTIZED_PATH = os.path.join(OUTPUT_DIR, "csl_encoder.onnx")  # 覆盖原文件


def convert():
    os.makedirs(OUTPUT_DIR, exist_ok=True)

    # 1. 加载 PyTorch 模型并导出为 ONNX
    print(f"Loading model: {MODEL_ID}")
    model = ORTModelForFeatureExtraction.from_pretrained(MODEL_ID, export=True)
    tokenizer = AutoTokenizer.from_pretrained(MODEL_ID)

    # 2. 保存 ONNX 模型与 tokenizer
    model.save_pretrained(OUTPUT_DIR)
    tokenizer.save_pretrained(OUTPUT_DIR)
    print(f"Saved ONNX model to: {OUTPUT_DIR}")

    # 3. INT8 动态量化
    raw_onnx = os.path.join(OUTPUT_DIR, "model.onnx")
    if os.path.exists(raw_onnx):
        print("Quantizing to INT8...")
        quantize_dynamic(
            model_input=raw_onnx,
            model_output=QUANTIZED_PATH,
            weight_type=QuantType.QInt8,
        )
        os.remove(raw_onnx)  # 删除未量化版本
        print(f"Done: {QUANTIZED_PATH}")
    else:
        # optimum 已直接保存为 model.onnx，重命名
        os.rename(raw_onnx, QUANTIZED_PATH)
        print(f"Saved (no quantization): {QUANTIZED_PATH}")


if __name__ == "__main__":
    convert()
```

### 2. FAISS 索引构建

```python
# scripts/build_faiss.py
"""
离线脚本：基于 ONNX 模型对标准词表进行向量化，构建 FAISS 索引。
运行环境：本地或云端均可（只需 ONNX 模型与词表）。
产出文件：data/faiss_index.bin, data/vocab.json
"""
import os
import json
import numpy as np
import faiss
from app.core.onnx_encoder import ONNXEncoder
from app.core.config import settings


VOCAB_SOURCE = "data/raw_vocab.txt"  # 每行一个标准手语词
VOCAB_OUTPUT = "data/vocab.json"
FAISS_OUTPUT = "data/faiss_index.bin"


def build():
    # 1. 加载词表
    with open(VOCAB_SOURCE, "r", encoding="utf-8") as f:
        words = [line.strip() for line in f if line.strip()]
    print(f"Loaded {len(words)} words")

    # 2. 保存词表 JSON
    with open(VOCAB_OUTPUT, "w", encoding="utf-8") as f:
        json.dump(words, f, ensure_ascii=False, indent=2)

    # 3. 加载 ONNX 编码器
    encoder = ONNXEncoder()
    encoder.load(settings.MODEL_PATH)

    # 4. 批量向量化
    embeddings = []
    batch_size = 32
    for i in range(0, len(words), batch_size):
        batch = words[i:i + batch_size]
        embs = encoder.encode(batch)
        embeddings.append(embs)
        print(f"Encoded {i + len(batch)}/{len(words)}")

    embeddings = np.vstack(embeddings).astype(np.float32)
    print(f"Embeddings shape: {embeddings.shape}")

    # 5. 构建 FAISS 索引（L2 距离，精确检索）
    dim = embeddings.shape[1]
    index = faiss.IndexFlatL2(dim)
    index.add(embeddings)
    print(f"FAISS index size: {index.ntotal}")

    # 6. 保存索引
    faiss.write_index(index, FAISS_OUTPUT)
    print(f"Saved: {FAISS_OUTPUT}")


if __name__ == "__main__":
    build()
```

### 3. AlignmentOp 强类型定义（与后端共享）

```python
# schemas/alignment.py
from pydantic import BaseModel, field_validator
from enum import Enum
from typing import Optional


class AlignmentOpType(str, Enum):
    POSTPONE = "postpone"    # 后置：word 移到 target 之后
    ADVANCE = "advance"      # 前置：word 移到 target 之前
    DELETE = "delete"        # 删除：word 从序列中移除
    INSERT = "insert"        # 插入：word 插入到 position 位置


class AlignmentOp(BaseModel):
    type: AlignmentOpType
    word: str
    target: Optional[str] = None    # postpone/advance 时必填
    position: Optional[int] = None  # insert 时必填

    @field_validator("target")
    @classmethod
    def validate_target(cls, v, info):
        if info.data.get("type") in (AlignmentOpType.POSTPONE, AlignmentOpType.ADVANCE):
            if not v:
                raise ValueError("postpone/advance 操作必须提供 target 字段")
        return v

    @field_validator("position")
    @classmethod
    def validate_position(cls, v, info):
        if info.data.get("type") == AlignmentOpType.INSERT:
            if v is None:
                raise ValueError("insert 操作必须提供 position 字段")
        return v
```

```python
# core/onnx_encoder.py
import numpy as np
import onnxruntime as ort
from transformers import AutoTokenizer
from app.core.config import settings
from loguru import logger


class ONNXEncoder:
    """ONNX 模型加载与向量化推理引擎。"""

    def __init__(self):
        self.session = None
        self.tokenizer = None

    def load(self, model_path: str):
        """加载 ONNX 模型与 tokenizer。"""
        # 加载 ONNX 会话
        self.session = ort.InferenceSession(
            model_path,
            providers=settings.ONNX_PROVIDERS,
        )
        logger.info(f"ONNX model loaded: {model_path}")

        # 加载 tokenizer（与 ONNX 模型同目录）
        tokenizer_path = "/".join(model_path.split("/")[:-1])
        self.tokenizer = AutoTokenizer.from_pretrained(tokenizer_path)
        logger.info(f"Tokenizer loaded from: {tokenizer_path}")

    def encode(self, texts: list[str]) -> np.ndarray:
        """将文本列表编码为向量（768 维）。"""
        if self.session is None or self.tokenizer is None:
            raise RuntimeError("ONNX encoder not loaded")

        # Tokenize
        inputs = self.tokenizer(
            texts,
            padding=True,
            truncation=True,
            max_length=128,
            return_tensors="np",
        )

        # ONNX 推理
        outputs = self.session.run(
            None,
            {
                "input_ids": inputs["input_ids"].astype(np.int64),
                "attention_mask": inputs["attention_mask"].astype(np.int64),
                "token_type_ids": inputs.get("token_type_ids", np.zeros_like(inputs["input_ids"])).astype(np.int64),
            },
        )

        # Mean Pooling（取 last_hidden_state 的均值）
        last_hidden_state = outputs[0]
        attention_mask = inputs["attention_mask"]
        mask_expanded = np.expand_dims(attention_mask, -1)
        sum_embeddings = np.sum(last_hidden_state * mask_expanded, axis=1)
        sum_mask = np.clip(mask_expanded.sum(axis=1), 1e-9, None)
        embeddings = sum_embeddings / sum_mask

        # L2 归一化
        norms = np.linalg.norm(embeddings, axis=1, keepdims=True)
        embeddings = embeddings / np.clip(norms, 1e-9, None)

        return embeddings.astype(np.float32)


# 全局单例
onnx_encoder = ONNXEncoder()
```

## 第五部分：正向改写流水线（NL → CSL）

### first_pass 透传契约（v3.3 补全，三组联调唯一真源）

> 1. **算法必须生成 `first_pass` 事件**。流水线 Step 1 分词 + OOV 检测后立即 yield `first_pass`，是流水线的固定动作，**不允许跳过**。即便消融实验 `ABLATION_SKIP_OOV=true`（走 jieba 快路径），仍要 yield `first_pass`，只是 `oov_status=false`。
> 2. **`first_pass` 永远不依赖上游开关**。与 `ENABLE_STREAM_MASKING` / `SHOW_OOV_MAP` / `SHOW_NMM_HINTS` 都无关。这三个开关由 BFF 读取并可能裁剪 `refined_pass` 字段，**不影响** `first_pass` 事件。
> 3. **后端必须透传 `first_pass` 事件**。`ChatService.stream_rewrite` 拿到的 `(event_name, event_data)` 原样 yield。
> 4. **前端负责表现**。UI 掩蔽全部在前端处理，不在算法这边。

### 流水线总览

```
输入: NL 文本 (如 "我吃苹果")
  ↓
Step 1: 分词 + OOV 检测 → first_pass_text, oov_status, oov_words
  ↓ (yield first_pass 事件)
Step 2: OOV 降级词解析 → oov_map
Step 3: 拦截器校验 → 是否需要 LLM 精炼
  ↓ (若不需要，跳到 Step 6)
Step 4: LLM 精炼 → refined_text
Step 5: 语序对齐指令生成 → alignment_ops (list[AlignmentOp]强类型)
Step 6: NMM 生成 → nmm_hints
Step 7: 线性化拼装 + OOV 队列投递 → refined_pass 数据
  ↓ (yield refined_pass 事件)
输出: refined_pass 事件
```

#### v3.3 alignment_ops 子结构与 JSON 示例（红线）

> **字段名须严格一致，严禁自行重命名。** 前后端依赖完全对齐才能驱动词卡飞行动画。

**AlignmentOp 子结构**（见 `schemas/alignment.py`）：

```python
class AlignmentOpType(str, Enum):
    POSTPONE = "postpone"    # 后置：word 移到 target 之后
    ADVANCE = "advance"      # 前置：word 移到 target 之前
    DELETE = "delete"        # 删除：word 从序列中移除
    INSERT = "insert"        # 插入：word 插入到 position 位置


class AlignmentOp(BaseModel):
    type: AlignmentOpType
    word: str
    target: Optional[str] = None    # postpone/advance 时必填
    position: Optional[int] = None  # insert 时必填
```

**refined_pass JSON 示例**（含 alignment_ops 子结构）：

```json
{
  "text": "东西 我 吃",
  "oov_map": {"苹果": "东西"},
  "nmm_hints": {"2": "NEGATION"},
  "alignment_ops": [
    {"type": "delete", "word": "的"},
    {"type": "postpone", "word": "苹果", "target": "我"},
    {"type": "insert", "word": "东西", "position": 0}
  ]
}
```

**Step 5 alignment_ops 生成规则**（`modules/alignment_generator.py`）：
- `type=delete`: 原句中要删的虚词（的、了、着、吗、呢、啊、吧）
- `type=postpone`: 宾语前置到主语之后（OSV 语序）
- `type=advance`: 状语/时间词前置
- `type=insert`: 补充原句没有但手语需要的词（如“东西”这种 OOV 降级词）
- 若生成列表为空（`[]`），表示原句与精炼句词序一致，仅作位置信息补充。

**v3.3 重要约定**：
- 字段名必须严格使用 `type/word/target/position` 四个键名
- `type` 必须是枚举值 `postpone/advance/delete/insert`，小写字符串
- `target` 仅在 `postpone/advance` 时填，其他类型填 `null`（序列化后省略）
- `position` 仅在 `insert` 时填，其他类型填 `null`
- 后端 BFF 透传时不做任何修改，前端 `AlignmentOp.fromJson` 会严格校验

### Step 1: 分词 + OOV 检测（v3.1 补全完整实现）

```python
# modules/oov_detector.py
import jieba
import numpy as np
import faiss
import json
from app.core.onnx_encoder import onnx_encoder
from app.core.config import settings
from loguru import logger


class OOVDetector:
    """分词 + FAISS 向量检索 OOV 检测。"""

    def __init__(self):
        self.faiss_index: faiss.Index = None
        self.vocab: list[str] = []

    def load(self, faiss_path: str, vocab_path: str):
        """加载 FAISS 索引与词表。"""
        self.faiss_index = faiss.read_index(faiss_path)
        with open(vocab_path, "r", encoding="utf-8") as f:
            self.vocab = json.load(f)
        logger.info(f"OOV detector loaded. FAISS size: {self.faiss_index.ntotal}, vocab size: {len(self.vocab)}")

    def detect(self, text: str) -> tuple[str, bool, list[str]]:
        """
        分词并检测 OOV。

        Args:
            text: 输入文本（原始自然语言，未分词）

        Returns:
            first_pass_text: 空格分词的文本（如 "我 吃 苹果"）
            oov_status: 是否检测到 OOV
            oov_words: OOV 词列表
        """
        # 1. jieba 分词
        tokens = list(jieba.cut(text))
        # 过滤空白 token
        tokens = [t for t in tokens if t.strip()]
        logger.debug(f"Tokenized: '{text}' -> {tokens}")

        # 2. 批量向量化所有 token
        embeddings = onnx_encoder.encode(tokens)

        # 3. FAISS 检索每个 token 的最近邻
        # v3.2 修复：换算公式说明。
        # FAISS IndexFlatL2 返回的是欧氏距离 d（0~+∞）。这里采用“似合度 = 1 / (1 + d)”换算为 0~1 区间的相似度：
        #   - d = 0   ->  similarity = 1.0 （完全一致）
        #   - d = 1   ->  similarity = 0.5
        #   - d = 4   ->  similarity = 0.2
        #   - d → +∞   ->  similarity → 0
        # OOV_SIMILARITY_THRESHOLD = 0.8 对应 d = 0.25，即余弦等价 0.8 左右的阈值。说明：
        #   -  ONNXEncoder.encode() 输出了 L2-normalized 向量（见 core/onnx_encoder.encode()），
        #      L2 距离与余弦距离可换算： cos_sim = 1 - d^2 / 2。
        #   -  当 ONNXEncoder 未做归一化时，这里可改为 similarity = 1.0 / (1.0 + distance / 2.0)。
        oov_words = []
        for i, token in enumerate(tokens):
            emb = np.array([embeddings[i]], dtype=np.float32)
            scores, indices = self.faiss_index.search(emb, 1)

            # FAISS 返回 L2 距离，转换为相似度
            distance = float(scores[0][0])
            similarity = 1.0 / (1.0 + distance)  # 距离越小相似度越高

            if similarity < settings.OOV_SIMILARITY_THRESHOLD:
                oov_words.append(token)
                logger.debug(f"OOV detected: '{token}' distance={distance:.4f} similarity={similarity:.3f}")

        first_pass_text = " ".join(tokens)
        oov_status = len(oov_words) > 0

        return first_pass_text, oov_status, oov_words


oov_detector = OOVDetector()
```

### Step 5: 语序对齐指令生成

```python
# modules/alignment_generator.py
from app.schemas.alignment import AlignmentOp, AlignmentOpType
from loguru import logger


class AlignmentGenerator:
    """生成从 original 到 refined 的对齐指令序列。

    基于 LCS（最长公共子序列）算法计算差异。
    """

    def generate(self, original: str, refined: str) -> list[AlignmentOp]:
        """
        生成对齐指令序列。

        Args:
            original: 原始文本（空格分词）
            refined: 精炼后文本（空格分词）

        Returns:
            AlignmentOp 列表
        """
        orig_tokens = original.split()
        refined_tokens = refined.split()

        ops = self._compute_diff(orig_tokens, refined_tokens)
        logger.debug(f"Alignment: {len(ops)} ops for '{original}' -> '{refined}'")
        return ops

    def _compute_diff(self, orig: list[str], refined: list[str]) -> list[AlignmentOp]:
        """基于 LCS 计算差异，生成对齐指令。"""
        m, n = len(orig), len(refined)

        # 计算 LCS 表
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if orig[i - 1] == refined[j - 1]:
                    dp[i][j] = dp[i - 1][j - 1] + 1
                else:
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])

        # 回溯生成指令
        ops = []
        i, j = m, n
        while i > 0 or j > 0:
            if i > 0 and j > 0 and orig[i - 1] == refined[j - 1]:
                # 相同，无需操作
                i -= 1
                j -= 1
            elif j > 0 and (i == 0 or dp[i][j - 1] >= dp[i - 1][j]):
                # refined 中有新词，需要 INSERT
                ops.append(AlignmentOp(
                    type=AlignmentOpType.INSERT,
                    word=refined[j - 1],
                    position=j - 1,
                ))
                j -= 1
            elif i > 0 and (j == 0 or dp[i - 1][j] > dp[i][j - 1]):
                # orig 中有词被删除
                ops.append(AlignmentOp(
                    type=AlignmentOpType.DELETE,
                    word=orig[i - 1],
                ))
                i -= 1

        # 检测 POSTPONE/ADVANCE（简化：若 refined 中词的相对顺序与 orig 不同）
        # 实际实现中，LCS 已覆盖大部分场景，POSTPONE/ADVANCE 为可选增强
        ops.reverse()
        return ops


alignment_generator = AlignmentGenerator()
```

```python
    1. 返回字符串标识（NEGATION/QUESTION/PAUSE），非 emoji。
    2. 前端映射：NEGATION→🚫, QUESTION→❓, PAUSE→⏸️。
    """

    # 否定词集合
    NEGATION_WORDS = {"不", "没", "无", "别", "勿"}

    # 疑问词集合
    QUESTION_WORDS = {"什么", "哪", "哪里", "哪儿", "谁", "怎么", "为什么", "吗", "呢"}

    # 停顿标记
    PAUSE_TOKEN = "<PAUSE>"

    def generate(self, refined_text: str) -> dict[str, str]:
        """
        生成 NMM 提示。

        Args:
            refined_text: 精炼后的 CSL 文本（空格分词）

        Returns:
            nmm_hints: {位置索引(字符串): 标识字符串}
                       标识取值：NEGATION / QUESTION / PAUSE
        """
        tokens = refined_text.split()
        nmm_hints = {}

        for i, token in enumerate(tokens):
            if token in self.NEGATION_WORDS:
                nmm_hints[str(i)] = "NEGATION"
            elif token in self.QUESTION_WORDS:
                nmm_hints[str(i)] = "QUESTION"
            elif token == self.PAUSE_TOKEN:
                nmm_hints[str(i)] = "PAUSE"

        if nmm_hints:
            logger.debug(f"NMM hints: {nmm_hints} for '{refined_text}'")
        return nmm_hints


nmm_generator = NMMGenerator()
```

```python
class Linearizer:
    """拼装最终 refined_pass 数据并投递 OOV 队列。

        refined_text: str,
        oov_map: dict[str, str],
        nmm_hints: dict[str, str],
        alignment_ops: list,

        拼装最终 refined_pass 数据并投递 OOV 队列。

        Args:
            refined_text: 精炼后的 CSL 文本

        Returns:
            refined_pass 事件的 data 字段（dict）
```

```python
# pipelines/forward_rewrite.py
import asyncio
import jieba  # v3.2 修复：上提至文件顶部，避免在条件内 import（Python 会缓存但仍不规范，且 lint 工具会报）。
from typing import AsyncGenerator, Tuple, Any
from app.core.config import settings
from app.core.logging import logger
from app.core.request_context import get_request_id
from app.modules.oov_detector import oov_detector
from app.modules.fallback_resolver import fallback_resolver
from app.modules.interceptor import should_refine
from app.modules.llm_refiner import llm_refiner
from app.modules.alignment_generator import alignment_generator
from app.modules.nmm_generator import nmm_generator
from app.modules.linearizer import linearizer


class ForwardRewritePipeline:
    """正向改写流水线：NL -> CSL。"""

    async def rewrite(self, text: str, context: str = None) -> AsyncGenerator[Tuple[str, Any], None]:
        """
        执行正向改写流水线，流式 yield SSE 事件。

        Args:
            text: 输入文本（自然语言）
            context: 上下文

        Yields:
            (event_name, event_data) 元组
            event_name: "first_pass" | "refined_pass"
            event_data: dict
        """
        rid = get_request_id()
        logger.info(f"Forward rewrite start: text='{text[:50]}' rid={rid}")

        # ===== Step 1: 分词 + OOV 检测 =====
        if settings.ABLATION_SKIP_OOV:
            # 消融实验：跳过 OOV 检测，直接 jieba 分词
            tokens = [t for t in jieba.cut(text) if t.strip()]
            first_pass_text = " ".join(tokens)
            oov_status = False
            oov_words = []
        else:
            first_pass_text, oov_status, oov_words = oov_detector.detect(text)

        logger.info(f"Step 1 done: first_pass='{first_pass_text}' oov_status={oov_status}")

        # ===== yield first_pass 事件（始终下发） =====
        yield "first_pass", {
            "text": first_pass_text,
            "oov_status": oov_status,
        }

        # ===== Step 2: OOV 降级词解析 =====
        oov_map = {}
        if oov_words:
            oov_map = await fallback_resolver.resolve(oov_words)
        logger.info(f"Step 2 done: oov_map={oov_map}")

        # ===== Step 3: 拦截器校验 =====
        need_refine = should_refine(first_pass_text, oov_status)
        logger.info(f"Step 3 done: need_refine={need_refine}")

        # ===== Step 4: LLM 精炼 =====
        if need_refine:
            refined_text = await llm_refiner.refine(first_pass_text, oov_map)
        else:
            # 跳过 LLM，直接用 first_pass + OOV 替换
            tokens = first_pass_text.split()
            refined_tokens = [oov_map.get(t, t) for t in tokens]
            refined_text = " ".join(refined_tokens)
        logger.info(f"Step 4 done: refined_text='{refined_text}'")

        # ===== Step 5: 语序对齐指令生成 =====
        alignment_ops = alignment_generator.generate(first_pass_text, refined_text)

        # ===== Step 6: NMM 生成 =====
        nmm_hints = {}
        if not settings.ABLATION_SKIP_NMM:
            nmm_hints = nmm_generator.generate(refined_text)

        # ===== Step 7: 线性化拼装 + OOV 队列投递 =====
        refined_data = await linearizer.assemble_and_dispatch(
            refined_text=refined_text,
            oov_map=oov_map,
            nmm_hints=nmm_hints,
            alignment_ops=alignment_ops,
        )

        # ===== yield refined_pass 事件 =====
        yield "refined_pass", refined_data
        logger.info(f"refined_pass yielded: text='{refined_text[:50]}'")


forward_rewrite_pipeline = ForwardRewritePipeline()
```

```python
# pipelines/reverse_normalize.py
from app.core.llm_client import deepseek_client
from app.core.logging import logger
from app.core.request_context import get_request_id


class ReverseNormalizePipeline:
    """逆向归一化流水线：CSL -> NL 候选。"""

    async def normalize(self, csl_text: str, context: str = None, num_options: int = 3) -> list[str]:
        """
        将 CSL 文本转换为多个 NL 候选。

        Args:
            csl_text: CSL 文本（空格分词）
            context: 上下文
            num_options: 候选数量（1-5）

        Returns:
            NL 候选列表
        """
        rid = get_request_id()
        logger.info(f"Reverse normalize start: text='{csl_text[:50]}' num={num_options} rid={rid}")

        prompt = f"""你是一个手语翻译专家。请将以下手语语序（CSL）文本转换为自然语言（NL），生成 {num_options} 条候选。

手语文本: {csl_text}
上下文: {context or "无"}

转换规则:
1. 将 OSV/SOV 语序转换为自然中文语序（SVO）
2. 补充必要的虚词（的、了、着等）
3. 每条候选不超过 30 字
4. 候选之间要有表达方式的变化（语气、句式）

返回 JSON: {{"options": ["候选1", "候选2", "候选3"]}}"""

        try:
            result = await deepseek_client.chat(
                messages=[{"role": "user", "content": prompt}],
                temperature=0.8,
            )
            options = result.get("options", [])

            # 确保返回指定数量的候选
            if len(options) < num_options:
                # 不足时补充默认候选
                options.append(csl_text.replace(" ", ""))
                while len(options) < num_options:
                    options.append(options[0])
            options = options[:num_options]

            logger.info(f"Reverse normalize done: {len(options)} options")
            return options

        except Exception as e:
            logger.error(f"Reverse normalize failed: {e}")
            # 回退：简单拼接
            fallback = csl_text.replace(" ", "")
            return [fallback] * num_options
```

### 1. 动态降级词典服务（v3.0 重构：使用 updated_at 时间戳；v3.2 修复连接复用）

```python
# services/dynamic_fallback_service.py
import httpx
from datetime import datetime
from app.core.config import settings
from app.core.logging import logger
from app.core.llm_client import get_shared_client  # v3.2 修复：复用共享 httpx client


class DynamicFallbackService:
    """动态降级词典管理（从后端拉取 + Redis PubSub 增量同步）。

    v3.0 重构：
    1. 废弃 version 列，改用 updated_at 时间戳进行增量同步。
    2. 响应体包含 server_time，算法引擎以此作为下次同步的 since 基准。
    v3.2 修复：
    1. 复用 app.core.llm_client 的共享 httpx.AsyncClient，避免每次拉取都新建连接。
    2. 区分 initial_load 与 incremental_sync 的超时（initial 10s 容忍冷启动，incremental 5s 紧缩）。
    """

    def __init__(self):
        self.dynamic_dict: dict[str, str] = {}  # {oov: fallback}
        self.last_sync_time: str = None  # ISO8601 时间戳

    async def initial_load(self):
        """启动时全量拉取。"""
        client = get_shared_client()
        resp = await client.get(
            f"http://{settings.BACKEND_HOST}:{settings.BACKEND_PORT}/internal/dynamic_fallback",
            timeout=10.0,
        )
        resp.raise_for_status()
        data = resp.json()
        for item in data["items"]:
            self.dynamic_dict[item["oov"]] = item["fallback"]
        self.last_sync_time = data.get("server_time")
        logger.info(
            f"Dynamic fallback loaded: {len(self.dynamic_dict)} items, sync_time={self.last_sync_time}"
        )

    async def incremental_sync(self):
        """增量同步（基于 updated_at 时间戳）。"""
        if not self.last_sync_time:
            # 首次同步失败，回退到全量
            await self.initial_load()
            return

        client = get_shared_client()
        resp = await client.get(
            f"http://{settings.BACKEND_HOST}:{settings.BACKEND_PORT}/internal/dynamic_fallback",
            params={"since": self.last_sync_time},
            timeout=5.0,
        )
        resp.raise_for_status()
        data = resp.json()
        new_count = 0
        for item in data["items"]:
            self.dynamic_dict[item["oov"]] = item["fallback"]
            new_count += 1
        # 更新同步时间戳
        self.last_sync_time = data.get("server_time", self.last_sync_time)

        if new_count > 0:
            logger.info(
                f"Dynamic fallback sync: +{new_count} new/updated, total={len(self.dynamic_dict)}"
            )

    def get(self, oov: str) -> str:
        """查询降级词。"""
        return self.dynamic_dict.get(oov)


dynamic_fallback_service = DynamicFallbackService()
```

### 2. Redis 服务（队列 + PubSub）（v3.2 修复 pubsub 实例化）

```python
# services/redis_service.py
import asyncio
import json
from redis.asyncio import Redis
from app.core.config import settings
from app.core.logging import logger
from app.services.dynamic_fallback_service import dynamic_fallback_service


class RedisService:
    """Redis 队列与 PubSub 服务。

    v3.2 修复：
    1. pubsub 由 redis_service 集中管理（self.pubsub），避免之前在 subscribe_dynamic_fallback
       函数内创建局部 pubsub 导致 redis_service.pubsub 始终为 None。
    2. close() 严格按 pubsub → client 顺序关闭，并加 try/except 防御。
    """

    def __init__(self):
        self.client: Redis = None
        self.pubsub = None  # 在 subscribe_dynamic_fallback 中赋值

    async def connect(self):
        """连接 Redis。"""
        self.client = Redis(
            host=settings.REDIS_HOST,
            port=settings.REDIS_PORT,
            db=settings.REDIS_DB,
            decode_responses=True,
            max_connections=5,
        )
        await self.client.ping()
        logger.info("Redis connected")

    async def close(self):
        """关闭连接。"""
        if self.pubsub is not None:
            try:
                await self.pubsub.close()
            except Exception as e:
                logger.warning(f"pubsub close failed: {e}")
            self.pubsub = None
        if self.client is not None:
            try:
                await self.client.close()
            except Exception as e:
                logger.warning(f"redis client close failed: {e}")
            self.client = None
        logger.info("Redis closed")

    def get_client(self) -> Redis:
        """获取已连接的 Redis 客户端，供 dynamic_fallback_service 等复用。"""
        if self.client is None:
            raise RuntimeError("RedisService not connected, call connect() first")
        return self.client

    async def lpush(self, key: str, value: str):
        """左推入队列。"""
        await self.client.lpush(key, value)

    async def blpop(self, key: str, timeout: int = 0):
        """阻塞左弹出。"""
        return await self.client.blpop(key, timeout=timeout)

    async def publish(self, channel: str, message: str):
        """发布消息。"""
        await self.client.publish(channel, message)


redis_service = RedisService()


async def subscribe_dynamic_fallback():
    """订阅动态词典更新通知。

    v3.2 修复：
    1. pubsub 赋值到 redis_service.pubsub，统一生命周期管理。
    2. listen() 循环外加 try/except，捕获异常后退避重连。
    """
    while True:  # 异常重连外层循环
        try:
            redis_service.pubsub = redis_service.client.pubsub()
            await redis_service.pubsub.subscribe("pubsub:dynamic_fallback_updated")
            logger.info("Subscribed to pubsub:dynamic_fallback_updated")
            async for message in redis_service.pubsub.listen():
                if message["type"] == "message":
                    logger.info("Received dynamic_fallback_updated notification")
                    await dynamic_fallback_service.incremental_sync()
        except asyncio.CancelledError:
            # lifespan 关闭时主动取消，正常退出
            raise
        except Exception as e:
            logger.exception(f"PubSub subscriber error, reconnecting in 3s: {e}")
            if redis_service.pubsub is not None:
                try:
                    await redis_service.pubsub.close()
                except Exception:
                    pass
                redis_service.pubsub = None
            await asyncio.sleep(3)


async def periodic_sync():
    """定时增量同步（每 10 分钟）。"""
    while True:
        await asyncio.sleep(settings.DYNAMIC_FALLBACK_SYNC_INTERVAL)
        logger.info("Periodic sync triggered")
        try:
            await dynamic_fallback_service.incremental_sync()
        except Exception as e:
            logger.exception(f"Periodic sync failed: {e}")
```

## 第十二部分：消融实验脚本

```python
# scripts/run_ablation.py
"""
消融实验脚本：验证各模块对整体性能的贡献。
指标：NPR（否定句保留率）、PNR（肯定句非反转率）。
"""
import asyncio
import json
import argparse
from app.core.config import settings
from app.pipelines.forward_rewrite import forward_rewrite_pipeline


async def run_test_case(text: str, expected: str) -> dict:
    """运行单个测试用例。"""
    events = []
    async for event_name, event_data in forward_rewrite_pipeline.rewrite(text, ""):
        events.append((event_name, event_data))

    refined_data = events[-1][1] if events else {}
    actual = refined_data.get("text", "")
    return {"input": text, "expected": expected, "actual": actual}


async def run_ablation(dataset_path: str, config_overrides: dict) -> list:
    """运行消融实验。

    v3.2 修复：使用 object.__setattr__ 绕过 pydantic v2 BaseSettings 的 __setattr__
    限制（某些版本默认为 frozen）。即使 Settings 设为 frozen=False，这里也安全。
    """
    # 应用配置覆盖
    for key, value in config_overrides.items():
        object.__setattr__(settings, key, value)

    with open(dataset_path, "r", encoding="utf-8") as f:
        test_cases = json.load(f)

    results = []
    for case in test_cases:
        result = await run_test_case(case["input"], case["expected"])
        results.append(result)
    return results


def calculate_npr(results: list) -> float:
    """计算否定句保留率。"""
    neg_results = [r for r in results if any(w in r["expected"] for w in ["不", "没", "无"])]
    if not neg_results:
        return 1.0
    preserved = sum(1 for r in neg_results if any(w in r["actual"] for w in ["不", "没", "无"]))
    return preserved / len(neg_results)


def calculate_pnr(results: list) -> float:
    """计算肯定句非反转率。"""
    pos_results = [r for r in results if not any(w in r["expected"] for w in ["不", "没", "无"])]
    if not pos_results:
        return 1.0
    not_reversed = sum(1 for r in pos_results if not any(w in r["actual"] for w in ["不", "没", "无"]))
    return not_reversed / len(pos_results)


async def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--dataset", default="data/test_set.json")
    args = parser.parse_args()

    configs = [
        ("full", {"ABLATION_SKIP_LLM": False, "ABLATION_SKIP_OOV": False, "ABLATION_SKIP_NMM": False}),
        ("no_llm", {"ABLATION_SKIP_LLM": True, "ABLATION_SKIP_OOV": False, "ABLATION_SKIP_NMM": False}),
        ("no_oov", {"ABLATION_SKIP_LLM": False, "ABLATION_SKIP_OOV": True, "ABLATION_SKIP_NMM": False}),
        ("no_nmm", {"ABLATION_SKIP_LLM": False, "ABLATION_SKIP_OOV": False, "ABLATION_SKIP_NMM": True}),
    ]

    all_results = {}
    for name, config in configs:
        print(f"\n===== Running ablation: {name} =====")
        results = await run_ablation(args.dataset, config)
        npr = calculate_npr(results)
        pnr = calculate_pnr(results)
        all_results[name] = {"npr": npr, "pnr": pnr, "results": results}
        print(f"NPR: {npr:.3f}, PNR: {pnr:.3f}")

    with open("data/ablation_results.json", "w", encoding="utf-8") as f:
        json.dump(all_results, f, ensure_ascii=False, indent=2)
    print("\nResults saved to data/ablation_results.json")


if __name__ == "__main__":
    asyncio.run(main())
```
