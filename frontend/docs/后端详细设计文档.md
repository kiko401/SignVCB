#### 3.2 `core/security.py`

JWT 签发与校验、密码哈希校验。

```python
from datetime import datetime, timedelta
from typing import Optional
from jose import jwt, JWTError
from passlib.context import CryptContext
from app.core.config import settings


pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def hash_password(password: str) -> str:
    return pwd_context.hash(password)


def verify_password(plain: str, hashed: str) -> bool:
    return pwd_context.verify(plain, hashed)


def create_access_token(subject: str) -> str:
    expire = datetime.utcnow() + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    payload = {"sub": subject, "exp": expire}
    return jwt.encode(payload, settings.JWT_SECRET, algorithm=settings.JWT_ALGORITHM)


def decode_access_token(token: str) -> Optional[str]:
    try:
        payload = jwt.decode(token, settings.JWT_SECRET, algorithms=[settings.JWT_ALGORITHM])
        return payload.get("sub")
    except JWTError:
        return None
```

```python
    port=settings.REDIS_PORT,
    db=settings.REDIS_DB,

async def get_db() -> AsyncSession:
    async with AsyncSessionLocal() as session:

            yield session

        except Exception:

        finally:
```

#### 3.4 `core/middleware.py`（工程红线）

请求追踪 ID 中间件。每个请求自动注入 `X-Request-ID`（若客户端未传则生成 UUID v4），存入 `contextvars` 供日志使用。

```python
import uuid
from contextvars import ContextVar
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import Response


request_id_ctx: ContextVar[str] = ContextVar("request_id", default="")


class RequestIDMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        request_id = request.headers.get("X-Request-ID") or str(uuid.uuid4())
        request_id_ctx.set(request_id)

        response: Response = await call_next(request)
        response.headers["X-Request-ID"] = request_id
        return response
```

#### 3.6 `core/exceptions.py`（工程红线）

全局异常捕获器。统一返回标准 JSON 错误体。

```python
from fastapi import Request, FastAPI
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from loguru import logger


class AppException(Exception):
    def __init__(self, code: str, message: str, status_code: int = 500):
        self.code = code
        self.message = message
        self.status_code = status_code


class AuthInvalidError(AppException):
    def __init__(self):
        super().__init__("AUTH_INVALID", "登录已失效，请重新登录", 401)


class RateLimitedError(AppException):
    def __init__(self):
        super().__init__("RATE_LIMITED", "操作太频繁了哦", 429)


class EngineTimeoutError(AppException):
    def __init__(self):
        super().__init__("ENGINE_TIMEOUT", "算法引擎响应超时", 502)


class EngineUnavailableError(AppException):
    def __init__(self):
        super().__init__("ENGINE_UNAVAILABLE", "算法引擎暂不可用", 503)


class ASRFailedError(AppException):
    def __init__(self, detail: str = "语音识别失败"):
        super().__init__("ASR_FAILED", detail, 500)


class ValidationError(AppException):
    def __init__(self, detail: str):
        super().__init__("VALIDATION_ERROR", detail, 422)


class InternalError(AppException):
    def __init__(self, detail: str = "服务器内部错误"):
        super().__init__("INTERNAL_ERROR", detail, 500)


def register_exception_handlers(app: FastAPI):
    @app.exception_handler(AppException)
    async def app_exception_handler(request: Request, exc: AppException):
        from app.core.middleware import request_id_ctx
        logger.warning(f"AppException: code={exc.code} message={exc.message}")
        return JSONResponse(
            status_code=exc.status_code,
            content={
                "code": exc.code,
                "message": exc.message,
                "request_id": request_id_ctx.get(),
            },
        )

    @app.exception_handler(RequestValidationError)
    async def validation_handler(request: Request, exc: RequestValidationError):
        from app.core.middleware import request_id_ctx
        errors = "; ".join([f"{'.'.join(str(x) for x in e['loc'])}: {e['msg']}" for e in exc.errors()])
        logger.warning(f"ValidationError: {errors}")
        return JSONResponse(
            status_code=422,
            content={
                "code": "VALIDATION_ERROR",
                "message": errors,
                "request_id": request_id_ctx.get(),
            },
        )

    @app.exception_handler(Exception)
    async def global_handler(request: Request, exc: Exception):
        from app.core.middleware import request_id_ctx
        logger.exception(f"Unhandled exception: {exc}")
        return JSONResponse(
            status_code=500,
            content={
                "code": "INTERNAL_ERROR",
                "message": "服务器内部错误",
                "request_id": request_id_ctx.get(),
            },
        )
```

### 5. AlignmentOp 强类型定义（与算法端共享）

```python
# schemas/alignment.py
from pydantic import BaseModel, field_validator
from enum import Enum
from typing import Optional


class AlignmentOpType(str, Enum):
    POSTPONE = "postpone"    # 后置：word 移到 target 之后
    ADVANCE = "advance"      # 前置：word 移到 target 之前
    DELETE = "delete"        # 删除：word 从序列中移除
    INSERT = "insert"        # 插入：word 插入到 position 位置


class AlignmentOp(BaseModel):
    type: AlignmentOpType
    word: str
    target: Optional[str] = None    # postpone/advance 时必填
    position: Optional[int] = None  # insert 时必填

    @field_validator("target")
    @classmethod
    def validate_target(cls, v, info):
        if info.data.get("type") in (AlignmentOpType.POSTPONE, AlignmentOpType.ADVANCE):
            if not v:
                raise ValueError("postpone/advance 操作必须提供 target 字段")
        return v

    @field_validator("position")
    @classmethod
    def validate_position(cls, v, info):
        if info.data.get("type") == AlignmentOpType.INSERT:
            if v is None:
                raise ValueError("insert 操作必须提供 position 字段")
        return v
```

### 7. SSE 事件契约（与前端状态机绝对对齐，v3.3 补强）

**first_pass 透传契约（v3.3 新增小节，作为三组联调唯一真源）**：

> 1. **算法必须生成 `first_pass` 事件**（yield first_pass 是流水线 Step 1 的固定动作，不允许跳过）。
> 2. **后端必须透传 `first_pass` 事件**（`ChatService.stream_rewrite` 拿到的 (event_name, event_data) 原样 yield，不允许过滤、不允许按 ENABLE_STREAM_MASKING 裁剪）。
> 3. **前端负责表现**：`ENABLE_STREAM_MASKING` 仅控制前端 UI 是否给 OOV 词卡加脉冲动画，不决定后端是否下发 `first_pass`。
> 4. **SHOW_OOV_MAP / SHOW_NMM_HINTS** 控制 `refined_pass` 中 `oov_map` / `nmm_hints` 字段是否填写（false 时算法填空字典，后端依然下发，只是前端应判断后跳过渲染）。

```python
# utils/sse.py
import json
from typing import Any


def format_sse_event(event: str, data: Any) -> str:
    """格式化 SSE 事件字符串。"""
    return f"event: {event}\ndata: {json.dumps(data, ensure_ascii=False)}\n\n"


# ===== SSE 事件类型与 data 字段定义（与前端状态机绝对对齐） =====
# v3.3 补强：data 字段的强类型定义在 schemas/chat.py 中（PreheatData / FirstPassData / RefinedPassData / FallbackData），
# 这里仅为代码注释参考，正式类型以前述 Pydantic 模型为准。
#
# preheat:      PreheatData(original="我吃苹果")
# first_pass:   FirstPassData(text="我 吃 东西", oov_status=True)   —— 始终下发，不受 ENABLE_STREAM_MASKING 影响
# refined_pass: RefinedData(text="东西 我 吃", oov_map={"苹果":"东西"}, nmm_hints={"2":"NEGATION"},
#                          alignment_ops=[AlignmentOp(type="postpone", word="苹果", target="东西")])
# fallback:     FallbackData(fallback_text="没听清哦")
```

> **注意**：`nmm_hints` 的值使用字符串标识而非 emoji，避免 JSON 编码问题。前端映射规则：`"NEGATION"` → 🚫，`"QUESTION"` → ❓，`"PAUSE"` → ⏸️。

## 第四部分：核心业务流程全链路 SOP（后端操作手册）

此部分指导后端工程师如何编排异步任务，确保与前端 SSE 状态机绝对对齐。

### 流程 1：语音输入 SSE 流式编排（`/api/v1/asr_and_rewrite`）

**触发场景**：用户长按麦克风录音后松手，前端上传 WAV 音频文件。

**实现步骤**：

1. **接收请求**：路由层接收 `multipart/form-data`，提取 `file` 和 `context`。
2. **下发 preheat**：立即向 SSE 流写入 `preheat` 事件，`original` 暂为空字符串（ASR 完成前）。
3. **调用 ASR**：`await asr_service.recognize(audio_bytes)` 获取转写文本。
4. **补发 preheat**：向 SSE 流写入第二条 `preheat` 事件，`original` 为 ASR 转写结果。
5. **调用算法引擎**：`async for event in engine_client.stream_rewrite(text, context, request_id)`。
6. **透传 first_pass**：收到算法引擎 `first_pass` 事件，原样透传给前端。
7. **透传 refined_pass**：收到算法引擎 `refined_pass` 事件，原样透传给前端。
8. **超时兜底**：若 `ENGINE_TIMEOUT`（10s）内未收到 `refined_pass`，向 SSE 流写入 `fallback` 事件。
9. **关闭流**：SSE 流正常结束。

**代码实现**：

```python
# api/v1/chat.py
import asyncio
import json
from fastapi import APIRouter, Depends, UploadFile, File, Form, Request
from fastapi.responses import StreamingResponse
from app.api.deps import get_current_user, limiter
from app.services.chat_service import ChatService
from app.utils.sse import format_sse_event
from app.core.exceptions import EngineTimeoutError, EngineUnavailableError
from app.schemas.chat import (
    RewriteRequest, TtsRequest, TtsResponse,
    SuggestReplyRequest, SuggestReplyResponse,
    NormalizeOptionsRequest, NormalizeOptionsResponse,
    LogMismatchRequest, LogMismatchResponse,
)
from loguru import logger

router = APIRouter()


@router.post("/asr_and_rewrite")
@limiter.limit("15/minute")
async def asr_and_rewrite(
    request: Request,
    file: UploadFile = File(...),
    context: str = Form(default=None),
    current_user=Depends(get_current_user),
):
    async def event_generator():
        try:
            # 1. 下发 preheat（original 暂空）
            yield format_sse_event("preheat", {"original": ""})

            # 2. 读取音频并调用 ASR
            audio_bytes = await file.read()
            try:
                text = await ChatService.asr_recognize(audio_bytes)
            except Exception as e:
                logger.error(f"ASR failed: {e}")
                yield format_sse_event("fallback", {"fallback_text": "没听清哦，再说一次好吗？"})
                return

            # 3. 补发 preheat（带 ASR 结果）
            yield format_sse_event("preheat", {"original": text})

            # 4. 调用算法引擎并透传事件
            async for event_name, event_data in ChatService.stream_rewrite(text, context, request.headers.get("X-Request-ID", "")):
                yield format_sse_event(event_name, event_data)

        except EngineTimeoutError:
            logger.warning("Engine timeout in asr_and_rewrite")
            yield format_sse_event("fallback", {"fallback_text": "网络有点慢哦，再试一次？"})
        except EngineUnavailableError:
            logger.error("Engine unavailable in asr_and_rewrite")
            yield format_sse_event("fallback", {"fallback_text": "服务暂不可用，稍后再试？"})
        except Exception as e:
            logger.exception(f"SSE stream error: {e}")
            yield format_sse_event("fallback", {"fallback_text": "出错了哦，再试一次？"})

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",  # Nginx 禁用缓冲
        },
    )
```

```python
        filepath = os.path.join(self.audio_dir, filename)

        # 缓存命中则直接返回
        if os.path.exists(filepath):
            logger.debug(f"TTS cache hit: {filename}")

        resp = await client.post(

            json={
                "model": "speech-01-turbo",
                "text": text,
                "speed": speed,
                "voice_setting": {"voice_id": "male-qn-qingse"},
            },
            timeout=10.0,
        )
        resp.raise_for_status()
        audio_data = base64.b64decode(resp.json()["data"]["audio"])
        with open(filepath, "wb") as f:
            f.write(audio_data)
        logger.info(f"TTS synthesized: {filename}, size={len(audio_data)}")
```

### 5. 算法引擎对接（v3.1 SSE 解析修复 + 共享连接池 + 懒加载 + 透传 request_id + 返回 dict）

```python
# services/engine_client.py
import httpx
import asyncio
import json
from typing import AsyncGenerator, Tuple, Any
from app.core.config import settings
from app.core.exceptions import EngineUnavailableError, EngineTimeoutError
from loguru import logger


class EngineClient:
    """算法引擎 HTTP 客户端（单例 + 懒加载 + 共享连接池）。

    v3.1 修复：
    1. 懒加载：首次调用时创建 httpx.AsyncClient，避免在事件循环启动前创建。
    2. 透传 X-Request-ID header。
    3. SSE 解析：event_data 在 yield 后重置为 None，避免复用上次数据。
    4. stream_rewrite 直接返回 dict（非 JSON 字符串），避免二次序列化。
    5. normalize 方法返回 list[str]。
    """

    _instance = None
    _client: httpx.AsyncClient = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    @classmethod
    def get_client(cls) -> httpx.AsyncClient:
        """懒加载共享 httpx 客户端。"""
        if cls._client is None or cls._client.is_closed:
            cls._client = httpx.AsyncClient(
                timeout=httpx.Timeout(settings.ENGINE_TIMEOUT, connect=settings.ENGINE_CONNECT_TIMEOUT),
                limits=httpx.Limits(max_connections=10, max_keepalive_connections=5),
            )
        return cls._client

    @classmethod
    async def close(cls):
        """关闭客户端（在 lifespan 关闭阶段调用）。"""
        if cls._client is not None and not cls._client.is_closed:
            await cls._client.aclose()
            cls._client = None

    @property
    def base_url(self) -> str:
        # v3.3 修复：优先使用 ALGORITHM_ENGINE_URL（包含 scheme/host/port），fallback 到 ENGINE_HOST:ENGINE_PORT。
        # 这样调整部署环境只需改一个变量。
        return settings.ALGORITHM_ENGINE_URL.rstrip("/") or f"http://{settings.ENGINE_HOST}:{settings.ENGINE_PORT}"

    async def stream_rewrite(self, text: str, context: str = None, request_id: str = "") -> AsyncGenerator[Tuple[str, Any], None]:
        """调用算法引擎 /internal/rewrite，流式返回 (event_name, event_data_dict)。

        v3.1 修复：
        1. 透传 X-Request-ID header 给算法引擎。
        2. event_data 在 yield 后重置为 None，避免复用上次数据。
        3. 共享 httpx.AsyncClient，复用 TCP 连接。
        4. 非 200 响应抛出 EngineUnavailableError。
        5. event_data 解析为 dict 后 yield，避免调用方二次 json.loads。
        """
        client = self.get_client()
        try:
            async with client.stream(
                "POST",
                f"{self.base_url}/internal/rewrite",
                json={"text": text, "context": context},
                headers={"X-Request-ID": request_id} if request_id else {},
            ) as response:
                if response.status_code != 200:
                    logger.error(f"Engine returned {response.status_code}")
                    raise EngineUnavailableError()

                event_name = None
                event_data = None
                async for line in response.aiter_lines():
                    if line.startswith("event:"):
                        event_name = line[6:].strip()
                    elif line.startswith("data:"):
                        event_data = line[5:].strip()
                        if event_name and event_data:
                            # 解析 JSON 字符串为 dict
                            try:
                                parsed_data = json.loads(event_data)
                            except json.JSONDecodeError:
                                logger.error(f"Failed to parse SSE data: {event_data}")
                                parsed_data = {}
                            yield event_name, parsed_data
                            # v3.1 修复：同时重置 event_name 和 event_data
                            event_name = None
                            event_data = None
        except httpx.TimeoutException:
            logger.warning(f"Engine stream timeout: {text[:50]}")
            raise EngineTimeoutError()
        except httpx.ConnectError:
            logger.error("Engine connection failed")
            raise EngineUnavailableError()

    async def normalize(self, text: str, context: str = None, num_options: int = 3) -> list[str]:
        """调用算法引擎 /internal/normalize 获取归一化候选。"""
        client = self.get_client()
        try:
            resp = await client.post(
                f"{self.base_url}/internal/normalize",
                json={"text": text, "context": context, "num_options": num_options},
                timeout=settings.ENGINE_TIMEOUT,
            )
            if resp.status_code != 200:
                logger.error(f"Engine normalize returned {resp.status_code}")
                raise EngineUnavailableError()
            return resp.json().get("options", [])
        except httpx.TimeoutException:
            logger.warning(f"Engine normalize timeout: {text[:50]}")
            raise EngineTimeoutError()
        except httpx.ConnectError:
            logger.error("Engine connection failed")
            raise EngineUnavailableError()

    async def health_check(self) -> bool:
        """健康检查。"""
        try:
            client = self.get_client()
            resp = await client.get(f"{self.base_url}/health", timeout=2.0)
            return resp.status_code == 200
        except Exception:
            return False
```

```python
# services/oov_consumer.py
import asyncio
import json
from sqlalchemy import text
from app.core.database import AsyncSessionLocal, redis_client
from app.core.config import settings
from loguru import logger


class OOVFallbackConsumer:
    """OOV 降级词典队列消费者。

    从 Redis 队列 queue:oov_fallback 拉取 OOV 词，
    UPSERT 到 dynamic_fallback 表，并发布 PubSub 通知算法引擎增量同步。
    """

    def __init__(self):
        self.queue_key = "queue:oov_fallback"
        self.pubsub_channel = "pubsub:dynamic_fallback_updated"

    async def start(self):
        """启动消费者（在 lifespan 中以 asyncio.create_task 调用）。"""
        logger.info("OOVFallbackConsumer started")
        while True:
            try:
                # BLPOP 阻塞读取，超时后重试
                result = await redis_client.blpop(
                    self.queue_key,
                    timeout=settings.OOV_CONSUMER_BLOCK_TIMEOUT,
                )
                if result is None:
                    continue

                _, message = result
                await self._process_message(message)

            except Exception as e:
                logger.exception(f"OOV consumer error: {e}")
                await asyncio.sleep(1)  # 异常后短暂等待

    async def _process_message(self, message: str):
        """处理单条消息：upsert + publish。"""
        try:
            data = json.loads(message)
            oov_word = data["oov"]
            fallback_word = data["fallback"]

            async with AsyncSessionLocal() as db:
                # UPSERT：存在则更新 fallback_word，不存在则插入
                # updated_at 由 MySQL ON UPDATE CURRENT_TIMESTAMP 自动更新
                await db.execute(
                    text("""
                        INSERT INTO dynamic_fallback (oov_word, fallback_word)
                        VALUES (:oov, :fallback)
                        ON DUPLICATE KEY UPDATE fallback_word = :fallback
                    """),
                    {"oov": oov_word, "fallback": fallback_word},
                )
                await db.commit()

            # 发布 PubSub 通知
            await redis_client.publish(
                self.pubsub_channel,
                json.dumps({"oov": oov_word, "fallback": fallback_word}),
            )
            logger.info(f"OOV consumed: {oov_word} -> {fallback_word}")

        except Exception as e:
            logger.exception(f"OOV process error: {e}")


oov_consumer = OOVFallbackConsumer()
```

```python
        q = result.scalar_one_or_none()

        if q is None:
            # 无可用题目（全部答过），回退到全量随机
            logger.info(f"No available questions for user={user_id}, fallback to all")
            stmt = (
                select(PracticeQuestion)
                .where(PracticeQuestion.level == level)
                .where(PracticeQuestion.type == type)
                .order_by(func.rand())
                .limit(1)
            )
            result = await db.execute(stmt)
            q = result.scalar_one()

        # v3.3 修复：拆开 question JSON 为强类型字段，不再返回开放的 dict。
        # 不同 type 的 question 字段不同，需分别取值。
        question_data = q.question or {}
        return PracticeQuestionResponse(

            level=q.level,
            type=q.type,

            image_urls=question_data.get('image_urls', []) or [],
            text=question_data.get('text'),

        if q is None:

            user_id=user_id, question_id=req.question_id, correct=correct

def _check_answer(user_answer, correct_answer) -> bool:

    if isinstance(user_answer, list) and isinstance(correct_answer, list):
        return sorted(user_answer) == sorted(correct_answer)
```

```python
from app.models.fallback import DynamicFallback
from loguru import logger

router = APIRouter()


@router.get("/internal/dynamic_fallback")
async def get_dynamic_fallback(since: str = Query(default=None)):
    """
    拉取动态降级词典。
    - since: ISO8601 时间戳，返回 updated_at > since 的记录。为空则全量返回。
    - 响应包含 server_time，算法引擎以此作为下次同步的 since 基准。
    """
    async with AsyncSessionLocal() as db:
        stmt = select(DynamicFallback)
        if since:
            try:
                since_dt = datetime.fromisoformat(since)
                stmt = stmt.where(DynamicFallback.updated_at > since_dt)
            except ValueError:
                logger.warning(f"Invalid since param: {since}")

        result = await db.execute(stmt)
        rows = result.scalars().all()

    return {
        "items": [
            {"oov": r.oov_word, "fallback": r.fallback_word}
            for r in rows
        ],
        "server_time": datetime.utcnow().isoformat(),
        "count": len(rows),
    }
```

```python
# api/v1/auth.py
from fastapi import APIRouter, Depends, Request
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from app.core.database import get_db
from app.api.deps import limiter
from app.core.security import hash_password, verify_password, create_access_token
from app.models.user import User
from app.schemas.auth import RegisterRequest, LoginRequest, TokenResponse, UserResponse
from app.core.exceptions import ValidationError

router = APIRouter()


@router.post("/register", response_model=TokenResponse)
@limiter.limit("5/minute")
async def register(req: RegisterRequest, request: Request, db: AsyncSession = Depends(get_db)):
    # 检查用户名是否已存在
    result = await db.execute(select(User).where(User.username == req.username))
    if result.scalar_one_or_none():
        raise ValidationError("用户名已存在")

    user = User(
        username=req.username,
        password_hash=hash_password(req.password),
        age_group=req.age_group,
        nickname=req.nickname,
    )
    db.add(user)
    await db.flush()

    token = create_access_token(str(user.id))
    return TokenResponse(
        access_token=token,
        user=UserResponse(
            id=user.id, username=user.username, age_group=user.age_group,
            nickname=user.nickname, avatar_url=user.avatar_url,
        ),
    )


@router.post("/login", response_model=TokenResponse)
@limiter.limit("5/minute")
async def login(req: LoginRequest, request: Request, db: AsyncSession = Depends(get_db)):
    result = await db.execute(select(User).where(User.username == req.username))
    user = result.scalar_one_or_none()
    if not user or not verify_password(req.password, user.password_hash):
        raise ValidationError("用户名或密码错误")

    token = create_access_token(str(user.id))
    return TokenResponse(
        access_token=token,
        user=UserResponse(
            id=user.id, username=user.username, age_group=user.age_group,
            nickname=user.nickname, avatar_url=user.avatar_url,
        ),
    )
```

### 0. 对外访问链路（v3.3 补全，三组联调唯一真源）

> **对外唯一入口**：`http://<SERVER_IP>:8081`（Nginx 监听）。**严禁**前端直连 8012 或 8000。

```

│ 客户端 (Flutter App / 浏览器)                                │
└──────────┬─────────────────────────────────────────────────┘
           │  HTTP 请求 (X-Request-ID, Authorization Bearer ...)
           ▼
┌────────────────────────────────────────────────────────────┐
│ 宿主机 Nginx :8081                                           │
│  /etc/nginx/conf.d/signvcb.conf                              │
│  location /api/  → proxy_pass http://127.0.0.1:8012          │
│  location ~^/api/v1/(asr_and_rewrite|rewrite) → 专用 SSE 配置  │

           ▼
┌────────────────────────────────────────────────────────────┐
│ Docker 宿主机 端口映射: 8012:8000                              │
│  signvcb-server 容器                                         │
│   - FastAPI :8000                                            │
│   - 后端 BFF (鉴权、限流、编排、透传、落库)                      │
└──────────┬─────────────────────────────────────────────────┘
           │  Docker 网络 signvcb-net
           ▼
┌────────────────────────────────────────────────────────────┐
│ signvcb-engine 容器 (不映射对外端口)                          │

│   - 通过容器名 signvcb-engine:8001 访问                       │
│   - ALGORITHM_ENGINE_URL=http://signvcb-engine:8001         │
└──────────┬─────────────────────────────────────────────────┘
           │
           ▼
┌────────────────────────────────────────────────────────────┐
│ 共享 MySQL 8.0 (signvcb-mysql:3306)  +  Redis 7 (signvcb-redis:6379) │
└────────────────────────────────────────────────────────────┘
```

**端口分配表（v3.3 补全）**：

| 链路点 | 地址 | 用途 | 谁访问 |
|:---|:---|:---|:---|
| 客户端 → Nginx | `<SERVER_IP>:8081` | 唯一公网入口 | 前端、测试脚本 |
| Nginx → signvcb-server | `127.0.0.1:8012` | 宿主机反代 | Nginx |
| signvcb-server 容器内 | `:8000` | FastAPI 监听 | Nginx 反代 |
| signvcb-server → signvcb-engine | `signvcb-engine:8001` | 跨容器调用 | 后端 EngineClient |
| signvcb-engine 容器内 | `:8001` | uvicorn 监听 | 后端 EngineClient |
| MySQL 宿主机映射 | `<SERVER_IP>:3308` | 运维 / DBeaver | DBA（按 IP 白名单） |

```nginx
upstream signvcb_backend {
```

## 第十三部分：验收清单

### 基建验收

- [ ] `docker-compose up -d` 四容器全部 `healthy`。
- [ ] `curl http://<IP>:8081/health` 返回 `{"status":"healthy","dependencies":{"mysql":"ok","redis":"ok","engine":"ok"}}`。

- [ ] `alembic upgrade head` 建表成功。
- [ ] `docker stats` 各容器内存不超配额，`free -h` 可用内存 ≥ 800MB。

### SSE 流验收

- [ ] 语音 SSE 请求事件流完整：preheat（空）→ preheat（ASR 结果）→ first_pass → refined_pass。
- [ ] 文本 SSE 请求事件流完整：preheat → first_pass → refined_pass。
- [ ] `first_pass` 事件始终透传，无论 `ENABLE_STREAM_MASKING` 开关。
- [ ] `alignment_ops` 字段为强类型 `AlignmentOp` 数组，`type` 为枚举值。
- [ ] `nmm_hints` 值为字符串标识（`NEGATION`/`QUESTION`/`PAUSE`），非 emoji。
- [ ] SSE 超时后下发 `fallback` 事件，前端显示 fallback 文案。
- [ ] `X-Request-ID` header 透传到算法引擎（算法端日志可查）。

### TTS 验收

- [ ] `curl -X POST http://<IP>:8081/api/v1/tts -H "Authorization: Bearer <token>" -d '{"text":"你好","speed":1.0}'` 返回 `audio_url`。
- [ ] `curl http://<IP>:8081/tts_audio/<filename>.mp3` 返回音频二进制流（非 404）。
- [ ] 相同文本第二次请求 TTS 命中缓存（日志出现 `TTS cache hit`）。

### OOV 闭环验收

- [ ] 触发 OOV 降级后，`SELECT * FROM dynamic_fallback` 有记录，`updated_at` 字段有值。
- [ ] 算法引擎日志出现 `incremental_sync: +1 items`，证明 PubSub 闭环生效。
- [ ] `docker exec signvcb-server redis-cli LLEN queue:oov_fallback` 队列长度为 0（消费者及时消费）。

### 归一化验收

- [ ] `curl -X POST http://<IP>:8081/api/v1/normalize_options -d '{"text":"苹果 我 吃"}'` 返回 3 条 NL 候选。
